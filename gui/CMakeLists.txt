cmake_minimum_required(VERSION 3.24)
project(NativeStitchedFlow LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.11.2 EXACT REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Multimedia Svg Test)
add_library(flow-host STATIC FlowHost.cpp FlowHost.h)
target_link_libraries(flow-host PUBLIC Qt6::Core Qt6::Gui Qt6::Multimedia)
target_include_directories(flow-host PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(flow-host PRIVATE -Wall -Wextra -Werror)
add_executable(native-stitched-flow main.cpp resources.qrc)
target_link_libraries(native-stitched-flow PRIVATE flow-host Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Svg)
add_executable(flow-tests tests/FlowTests.cpp resources.qrc)
target_link_libraries(flow-tests PRIVATE flow-host Qt6::Test Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Svg)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME flow-validation COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/validate.py
    --bundle ${CMAKE_BINARY_DIR}/../bundle --tests $<TARGET_FILE:flow-tests>
    --output ${CMAKE_BINARY_DIR}/validation-cohort)
foreach(target native-stitched-flow flow-tests)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
endforeach()
